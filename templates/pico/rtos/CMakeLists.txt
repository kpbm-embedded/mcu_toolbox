cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
# FreeRTOS-Kernel with the RP2040 / RP2350 ports: -DFREERTOS_KERNEL_PATH=..., or $FREERTOS_KERNEL_PATH,
# or $MCU_ROOT/freertos/FreeRTOS-Kernel (MCU_ROOT = the mcu tool dir).
if (NOT FREERTOS_KERNEL_PATH AND NOT DEFINED ENV{FREERTOS_KERNEL_PATH})
    if (DEFINED ENV{MCU_ROOT} AND EXISTS "$ENV{MCU_ROOT}/freertos/FreeRTOS-Kernel/tasks.c")
        set(FREERTOS_KERNEL_PATH "$ENV{MCU_ROOT}/freertos/FreeRTOS-Kernel")
    else()
        message(FATAL_ERROR "Set FREERTOS_KERNEL_PATH (env or -D) or MCU_ROOT to locate FreeRTOS-Kernel")
    endif()
endif()
project({{NAME}} C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
include(FreeRTOS_Kernel_import.cmake)
add_executable({{NAME}} main.c)
target_include_directories({{NAME}} PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries({{NAME}} pico_stdlib FreeRTOS-Kernel-Heap4)
pico_enable_stdio_usb({{NAME}} 0)
pico_enable_stdio_uart({{NAME}} 1)
pico_add_extra_outputs({{NAME}})
