cmake_minimum_required(VERSION 3.20)
# Needs: . mcu/freertos/env.sh   (Arm GNU toolchain, cmake, ninja, STM32 Cube path, FREERTOS_KERNEL)
if(NOT DEFINED ENV{STM32_CUBE_F4} OR NOT DEFINED ENV{FREERTOS_KERNEL})
  message(FATAL_ERROR "source mcu/freertos/env.sh first")
endif()
set(STM32_BOARDS "$ENV{STM32_CUBE_F4}/../boards" CACHE PATH "mcu/stm32/boards (shared board layer)")
set(CMAKE_TOOLCHAIN_FILE ${STM32_BOARDS}/arm-none-eabi.cmake)
set(BOARD "{{TARGET}}" CACHE STRING "blackpill-f411 | weact-g431 | mini-h743 | bluepill-f103")
project({{NAME}} C ASM)
include(${STM32_BOARDS}/board.cmake)
include($ENV{FREERTOS_KERNEL}/../common/freertos.cmake)
add_executable({{NAME}} src/main.c)
freertos_board({{NAME}})
