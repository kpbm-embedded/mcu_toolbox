cmake_minimum_required(VERSION 3.20)
# Needs: . mcu/stm32/env.sh   (Arm GNU toolchain, cmake, ninja, STM32_CUBE_* paths)
if(NOT DEFINED ENV{STM32_CUBE_F4})
  message(FATAL_ERROR "source mcu/stm32/env.sh first")
endif()
set(STM32_BOARDS "$ENV{STM32_CUBE_F4}/../boards" CACHE PATH "mcu/stm32/boards (shared board layer)")
set(CMAKE_TOOLCHAIN_FILE ${STM32_BOARDS}/arm-none-eabi.cmake)
set(BOARD "blackpill-f411" CACHE STRING "blackpill-f411 | weact-g431 | mini-h743 | bluepill-f103")
project(usb-cdc C ASM)
include(${STM32_BOARDS}/board.cmake)
add_executable(usb-cdc src/main.c)
stm32_board(usb-cdc MODULES pcd DEFINES BOARD_USB48=1)
# USB Device middleware from the Cube pack of the board family (read-only, built out of tree)
set(USBD $ENV{STM32_CUBE_${B_FAMILY}}/Middlewares/ST/STM32_USB_Device_Library)
target_sources(usb-cdc PRIVATE src/usbd_conf.c src/usbd_desc.c ${USBD}/Core/Src/usbd_core.c ${USBD}/Core/Src/usbd_ctlreq.c
  ${USBD}/Core/Src/usbd_ioreq.c ${USBD}/Class/CDC/Src/usbd_cdc.c)
target_include_directories(usb-cdc PRIVATE src ${USBD}/Core/Inc ${USBD}/Class/CDC/Inc)
