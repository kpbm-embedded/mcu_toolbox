cmake_minimum_required(VERSION 3.5)
# Out-of-tree project: cmake -S . -B build -DSDK=<kendryte-freertos-sdk> -DTOOLCHAIN=<toolchain/bin> -DBOARD=m1n|cm1 -DCMAKE_POLICY_VERSION_MINIMUM=3.5
if(NOT SDK)
  message(FATAL_ERROR "pass -DSDK=/path/to/kendryte-freertos-sdk")
endif()
include(${SDK}/cmake/common.cmake)
include(${SDK}/cmake/macros.internal.cmake OPTIONAL)
project(rtos-tasks C CXX ASM)
if(NOT BOARD)
  set(BOARD m1n)
endif()
string(TOUPPER ${BOARD} BOARD_UC)
add_definitions(-DBOARD_${BOARD_UC}=1)
add_source_files(main.c)
# mirror the SDK's own top-level CMakeLists.txt: lib + third_party (lwip, fatfs, gsl) built here, out of tree
include_directories(${SDK}/lib/arch/include ${SDK}/lib/utils/include ${SDK}/lib/hal/include)
header_directories(${SDK}/lib)
add_subdirectory(${SDK}/lib SDK)
add_subdirectory(${SDK}/third_party third_party)
set(BUILDING_SDK yes)
include(${SDK}/cmake/executable.cmake)
